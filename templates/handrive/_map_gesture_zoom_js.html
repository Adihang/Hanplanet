function enableMapGestureZoom(map, container) { if (!map || !container) return; if (map.touchZoom && map.touchZoom.disable) { map.touchZoom.disable(); } if (map.scrollWheelZoom && map.scrollWheelZoom.enable) { map.scrollWheelZoom.enable(); } if (map.doubleClickZoom && map.doubleClickZoom.enable) { map.doubleClickZoom.enable(); } var touches = new Map(); var pinch = { active: false, startDistance: 0, startZoom: 0 }; function touchMidpoint() { var points = Array.from(touches.values()); if (!points.length) return null; var x = 0; var y = 0; points.forEach(function(point) { x += point.x; y += point.y; }); return { x: x / points.length, y: y / points.length }; } function touchDistance() { var points = Array.from(touches.values()); if (points.length < 2) return 0; var dx = points[0].x - points[1].x; var dy = points[0].y - points[1].y; return Math.sqrt(dx * dx + dy * dy); } function setDragging(enabled) { if (!map.dragging) return; if (enabled) { if (map.dragging.enable) map.dragging.enable(); } else { if (map.dragging.disable) map.dragging.disable(); } } function onTouchStart(event) { if (!event || !event.touches) return; for (var i = 0; i < event.touches.length; i += 1) { var touch = event.touches[i]; touches.set(touch.identifier, { x: touch.clientX, y: touch.clientY }); } if (event.touches.length >= 2) { pinch.active = true; pinch.startDistance = touchDistance() || 1; pinch.startZoom = map.getZoom(); setDragging(false); event.preventDefault(); event.stopPropagation(); } } function onTouchMove(event) { if (!event || !event.touches) return; for (var i = 0; i < event.touches.length; i += 1) { var touch = event.touches[i]; touches.set(touch.identifier, { x: touch.clientX, y: touch.clientY }); } if (!pinch.active || event.touches.length < 2) return; event.preventDefault(); event.stopPropagation(); var currentDistance = touchDistance() || pinch.startDistance || 1; var ratio = currentDistance / (pinch.startDistance || currentDistance); var nextZoom = pinch.startZoom + Math.log(ratio) / Math.LN2; var mid = touchMidpoint(); if (!mid) return; var latlng = map.containerPointToLatLng([mid.x, mid.y]); map.setZoomAround(latlng, nextZoom, { animate: false }); } function onTouchEnd(event) { if (!event || !event.changedTouches) return; for (var i = 0; i < event.changedTouches.length; i += 1) { var touch = event.changedTouches[i]; touches.delete(touch.identifier); } if (pinch.active && touches.size < 2) { pinch.active = false; pinch.startDistance = 0; setDragging(true); } } function zoomFromWheel(event) { if (!event || !(event.ctrlKey || event.metaKey)) return; event.preventDefault(); event.stopPropagation(); var point = map.mouseEventToContainerPoint(event); var delta = event.deltaY || 0; var nextZoom = map.getZoom() - (delta / 240) * 0.8; map.setZoomAround(point, nextZoom, { animate: true, duration: 0.16 }); } container.addEventListener("touchstart", onTouchStart, { passive: false, capture: true }); container.addEventListener("touchmove", onTouchMove, { passive: false, capture: true }); container.addEventListener("touchend", onTouchEnd, { passive: false, capture: true }); container.addEventListener("touchcancel", onTouchEnd, { passive: false, capture: true }); container.addEventListener("wheel", zoomFromWheel, { capture: true, passive: false }); }
