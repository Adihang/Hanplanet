function enableMapGestureZoom(map, container) { if (!map || !container) return; if (map.touchZoom && map.touchZoom.disable) { map.touchZoom.disable(); } // Leaflet's built-in wheel handler debounces the stream and starts a // separate zoom animation for each burst. On high-resolution wheels // this drops intermediate deltas, making a long scroll feel like one // zoom step at a time. Keep the handler disabled and consume the stream // below in one rAF-driven target animation instead. if (map.scrollWheelZoom && map.scrollWheelZoom.disable) { map.scrollWheelZoom.disable(); } if (map.doubleClickZoom && map.doubleClickZoom.enable) { map.doubleClickZoom.enable(); } var touches = new Map(); var pinch = { active: false, startDistance: 0, startZoom: 0 }; function touchMidpoint() { var points = Array.from(touches.values()); if (!points.length) return null; var x = 0; var y = 0; points.forEach(function(point) { x += point.x; y += point.y; }); return { x: x / points.length, y: y / points.length }; } function touchDistance() { var points = Array.from(touches.values()); if (points.length < 2) return 0; var dx = points[0].x - points[1].x; var dy = points[0].y - points[1].y; return Math.sqrt(dx * dx + dy * dy); } function setDragging(enabled) { if (!map.dragging) return; if (enabled) { if (map.dragging.enable) map.dragging.enable(); } else { if (map.dragging.disable) map.dragging.disable(); } } function onTouchStart(event) { if (!event || !event.touches) return; for (var i = 0; i < event.touches.length; i += 1) { var touch = event.touches[i]; touches.set(touch.identifier, { x: touch.clientX, y: touch.clientY }); } if (event.touches.length >= 2) { pinch.active = true; pinch.startDistance = touchDistance() || 1; pinch.startZoom = map.getZoom(); setDragging(false); event.preventDefault(); event.stopPropagation(); } } function onTouchMove(event) { if (!event || !event.touches) return; for (var i = 0; i < event.touches.length; i += 1) { var touch = event.touches[i]; touches.set(touch.identifier, { x: touch.clientX, y: touch.clientY }); } if (!pinch.active || event.touches.length < 2) return; event.preventDefault(); event.stopPropagation(); var currentDistance = touchDistance() || pinch.startDistance || 1; var ratio = currentDistance / (pinch.startDistance || currentDistance); var nextZoom = pinch.startZoom + Math.log(ratio) / Math.LN2; var mid = touchMidpoint(); if (!mid) return; var latlng = map.containerPointToLatLng([mid.x, mid.y]); map.setZoomAround(latlng, nextZoom, { animate: false }); } function onTouchEnd(event) { if (!event || !event.changedTouches) return; for (var i = 0; i < event.changedTouches.length; i += 1) { var touch = event.changedTouches[i]; touches.delete(touch.identifier); } if (pinch.active && touches.size < 2) { pinch.active = false; pinch.startDistance = 0; setDragging(true); } } var wheelState = { pendingDelta: 0, targetZoom: null, point: null, frame: 0, }; function isImageAdjusting() { var editorRoot = document.getElementById("map-editor-root"); return !!(editorRoot && editorRoot.classList.contains("is-map-image-adjusting")); } function isWheelControlTarget(event) { var target = event && event.target; return !!(target && target.closest && target.closest( ".leaflet-control, .leaflet-popup, .map-viewer-layers-widget" )); } function normalizeWheelDelta(event) { var delta = Number(event && event.deltaY) || 0; if (Math.abs(delta) <= 0.01) { delta = Number(event && event.deltaX) || 0; } if (event && event.deltaMode === 1) { delta *= 16; } else if (event && event.deltaMode === 2) { delta *= window.innerHeight || 800; } return delta; } function flushWheelZoom() { wheelState.frame = 0; if (!map || wheelState.targetZoom === null) return; if (wheelState.pendingDelta) { // Match HanDrive's shared zoom gesture: 100 normalized wheel // pixels equals one 1.125x scale step. Leaflet zoom is logarithmic // (one unit doubles the scale), so convert the same ratio to map // zoom units before applying it. var wheelRatio = 1.125; var wheelDeltaUnit = 100; var scaleRatio = Math.pow(wheelRatio, -(wheelState.pendingDelta / wheelDeltaUnit)); var mapZoomDelta = Math.log(scaleRatio) / Math.LN2; wheelState.targetZoom = map._limitZoom(wheelState.targetZoom + mapZoomDelta); wheelState.pendingDelta = 0; } var currentZoom = map.getZoom(); var difference = wheelState.targetZoom - currentZoom; if (Math.abs(difference) < 0.001) { map.setZoomAround(wheelState.point || map.getSize().divideBy(2), wheelState.targetZoom, { animate: false, }); wheelState.targetZoom = map.getZoom(); return; } // Advance a fraction of the remaining distance so consecutive wheel // events are rendered smoothly instead of as isolated jumps. var nextZoom = currentZoom + difference * 0.32; if (map._stop) map._stop(); map.setZoomAround(wheelState.point || map.getSize().divideBy(2), nextZoom, { animate: false, }); wheelState.frame = window.requestAnimationFrame(flushWheelZoom); } function zoomFromWheel(event) { if (!event || isImageAdjusting() || isWheelControlTarget(event)) return; var delta = normalizeWheelDelta(event); if (!delta) return; event.preventDefault(); event.stopPropagation(); wheelState.pendingDelta += delta; wheelState.point = map.mouseEventToContainerPoint(event); if (wheelState.targetZoom === null) { wheelState.targetZoom = map.getZoom(); } if (!wheelState.frame) { wheelState.frame = window.requestAnimationFrame(flushWheelZoom); } } container.addEventListener("touchstart", onTouchStart, { passive: false, capture: true }); container.addEventListener("touchmove", onTouchMove, { passive: false, capture: true }); container.addEventListener("touchend", onTouchEnd, { passive: false, capture: true }); container.addEventListener("touchcancel", onTouchEnd, { passive: false, capture: true }); container.addEventListener("wheel", zoomFromWheel, { capture: true, passive: false }); }
